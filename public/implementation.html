Canonical source for the code excerpts used in the story. This educational implementation uses random initial weights, a 256-token context, and an untied output projection. It does not load a trained GPT-2 checkpoint.

Follow the implementation

  2. Prepare examplesText splits and shifted context-target windows
  3. Represent the sentenceToken and position embeddings, repeated blocks, final output scores
  4. Connect positionsQuery, key, and value projections; causal attention
  5. Transform each positionExpansion, GELU, and projection back to model width
  6. Repeat the blockPre-normalized attention and feed-forward residual paths
  7. Continue the sentenceTemperature, optional top-k, selection, append, and decode
  8. How learning worksCross-entropy, gradients, optimizer steps, and evaluation

20def read_file(file_path, train_split_ratio):
21    with open(file_path, "r", encoding="utf-8") as file:
22        text = file.read()
23    split_at_this_index = int(train_split_ratio * len(text))
24    train_text = text[:split_at_this_index]
25    validation_text = text[split_at_this_index:]
26    return train_text, validation_text

29class SlidingWindowDataset(Dataset):
30    def __init__(self, text, tokenizer, context_length, stride):
31        self.source_tokens = []
32        self.target_tokens = []
33        tokens = tokenizer.encode(text)
34        for token_id in range(0, len(tokens) - context_length, stride):
35            source = tokens[token_id:token_id + context_length]
36            target = tokens[token_id + 1:token_id + context_length + 1]
37            self.source_tokens.append(torch.tensor(source))
38            self.target_tokens.append(torch.tensor(target))
39 
40    def __len__(self):
41        return len(self.source_tokens)
42 
43    def __getitem__(self, index):
44        return self.source_tokens[index], self.target_tokens[index]

54# region feedforward
55class FeedForward(nn.Module):
56    def __init__(self, config):
57        super().__init__()
58        self.layers = nn.Sequential(
59            nn.Linear(config["emb_dim"], 4 * config["emb_dim"]),
60            nn.GELU(),
61            nn.Linear(4 * config["emb_dim"], config["emb_dim"]),
62        )
63 
64    def forward(self, x):
65        return self.layers(x)

68# region attention
69class MultiHeadAttention(nn.Module):
70    def __init__(self, dim_in, dim_out, context_length,
71                 dropout, num_heads, qkv_bias=False):
72        super().__init__()
73        assert dim_out % num_heads == 0
74        self.dim_out = dim_out
75        self.num_heads = num_heads
76        self.head_dim = dim_out // num_heads
77 
78        self.Wq = nn.Linear(dim_in, dim_out, bias=qkv_bias)
79        self.Wk = nn.Linear(dim_in, dim_out, bias=qkv_bias)
80        self.Wv = nn.Linear(dim_in, dim_out, bias=qkv_bias)
81        self.head_combined_projection = nn.Linear(dim_out, dim_out)
82        self.dropout = nn.Dropout(dropout)
83        self.register_buffer(
84            "mask", torch.triu(torch.ones(context_length, context_length), diagonal=1)
85        )
86 
87    def forward(self, x):
88        batch, tokens, _ = x.shape
89        q = self.Wq(x)
90        k = self.Wk(x)
91        v = self.Wv(x)
92 
93        q = q.view(batch, tokens, self.num_heads, self.head_dim).transpose(1, 2)
94        k = k.view(batch, tokens, self.num_heads, self.head_dim).transpose(1, 2)
95        v = v.view(batch, tokens, self.num_heads, self.head_dim).transpose(1, 2)
96 
97        scores = q @ k.transpose(2, 3)
98        mask = self.mask.bool()[:tokens, :tokens]
99        scores.masked_fill_(mask, -torch.inf)
100        weights = torch.softmax(scores / self.head_dim ** 0.5, dim=-1)
101        weights = self.dropout(weights)
102 
103        context = (weights @ v).transpose(1, 2).contiguous()
104        context = context.view(batch, tokens, self.dim_out)
105        return self.head_combined_projection(context)
106# endregion attention

108# region block
109class TransformerBlock(nn.Module):
110    def __init__(self, config):
111        super().__init__()
112        self.attention = MultiHeadAttention(
113            dim_in=config["emb_dim"], dim_out=config["emb_dim"],
114            context_length=config["context_length"],
115            num_heads=config["n_heads"], dropout=config["drop_rate"],
116            qkv_bias=config["qkv_bias"],
117        )
118        self.feedforwards = FeedForward(config)
119        self.layer_norm_1 = nn.LayerNorm(config["emb_dim"])
120        self.layer_norm_2 = nn.LayerNorm(config["emb_dim"])
121        self.drop_shortcut = nn.Dropout(config["drop_rate"])
122 
123    def forward(self, x):
124        shortcut = x
125        x = self.layer_norm_1(x)
126        x = self.drop_shortcut(self.attention(x))
127        x = x + shortcut
128 
129        shortcut = x
130        x = self.layer_norm_2(x)
131        x = self.drop_shortcut(self.feedforwards(x))
132        return x + shortcut
133# endregion block

135# region model
136class GPT(nn.Module):
137    def __init__(self, config):
138        super().__init__()
139        self.config = config
140        self.tokens_embeddings = nn.Embedding(config["vocab_size"], config["emb_dim"])
141        self.positional_embeddings = nn.Embedding(config["context_length"], config["emb_dim"])
142        self.dropout = nn.Dropout(config["drop_rate"])
143        self.transformer = nn.Sequential(
144            *(TransformerBlock(config) for _ in range(config["n_layers"]))
145        )
146        self.final_norm = nn.LayerNorm(config["emb_dim"])
147        # This teaching implementation keeps the output projection untied; GPT-2 ties it.
148        self.output_head = nn.Linear(config["emb_dim"], config["vocab_size"], bias=False)
149 
150    def forward(self, idx):
151        _, seq_len = idx.shape
152        tok_emb = self.tokens_embeddings(idx)
153        positions = torch.arange(seq_len, device=idx.device)
154        pos_emb = self.positional_embeddings(positions)
155        x = self.dropout(tok_emb + pos_emb)
156        x = self.final_norm(self.transformer(x))
157        return self.output_head(x)

171# region generation
172def generate(model, idx, max_generated_tokens, temperature=1.0, top_k=None):
173    if temperature < 0:
174        raise ValueError("temperature must be zero (greedy) or positive")
175 
176    for _ in range(max_generated_tokens):
177        idx_context = idx[:, -model.config["context_length"]:]
178        with torch.no_grad():
179            logits = model(idx_context)[:, -1, :]
180 
181        if temperature == 0:
182            idx_next = torch.argmax(logits, dim=-1, keepdim=True)
183        else:
184            logits = logits / temperature
185            if top_k is not None:
186                if not 1 <= top_k <= logits.size(-1):
187                    raise ValueError("top_k must be between 1 and vocabulary size")
188                top_logits, _ = torch.topk(logits, top_k)
189                cutoff = top_logits[:, -1, None]
190                logits = logits.masked_fill(logits < cutoff, -torch.inf)
191            probs = torch.softmax(logits, dim=-1)
192            idx_next = torch.multinomial(probs, num_samples=1)
193        idx = torch.cat((idx, idx_next), dim=1)
194    return idx

205# region training
206def get_loss_per_batch(input_batch, target_batch, model, device):
207    input_batch = input_batch.to(device)
208    target_batch = target_batch.to(device)
209    logits = model(input_batch)
210    return F.cross_entropy(logits.flatten(0, 1), target_batch.flatten())

213@torch.no_grad()
214def get_loss_per_dataloader(dataloader, model, device, num_batches=None):
215    total_loss = 0
216    num_batches = min(num_batches or len(dataloader), len(dataloader))
217    for i, (x, y) in enumerate(dataloader):
218        if i >= num_batches:
219            break
220        total_loss += get_loss_per_batch(x, y, model, device).item()
221    return total_loss / num_batches if num_batches > 0 else float("nan")

224def train_model(model, train_dl, val_dl, optimizer, device,
225                num_epochs, eval_freq, eval_iter):
226    train_losses, val_losses, track_tokens_seen = [], [], []
227    tokens_seen, step = 0, -1
228    for epoch in range(num_epochs):
229        model.train()
230        for x, y in train_dl:
231            optimizer.zero_grad()
232            loss = get_loss_per_batch(x, y, model, device)
233            loss.backward()
234            optimizer.step()
235            tokens_seen += x.numel()
236            step += 1
237 
238            if step % eval_freq == 0:
239                model.eval()
240                train_loss = get_loss_per_dataloader(train_dl, model, device, eval_iter)
241                val_loss = get_loss_per_dataloader(val_dl, model, device, eval_iter)
242                train_losses.append(train_loss)
243                val_losses.append(val_loss)
244                track_tokens_seen.append(tokens_seen)
245                print(f"Epoch {epoch + 1} | Step {step:06d} | "
246                      f"Train {train_loss:.3f} | Val {val_loss:.3f}")
247                model.train()
248    return train_losses, val_losses, track_tokens_seen
